1. Aim & Objectives

Aim: To design, develop, and deploy a RESTful Application Programming Interface (API) using Node.js and Express.js, connect it to a database (MongoDB / Supabase), implement full Create, Read, Update, and Delete (CRUD) operations, deploy the service to a cloud platform (Render), and verify all endpoints using Postman.

  • Design clean REST resource endpoints following standard HTTP methods (POST, GET, PUT, DELETE).
  • Connect the backend application to a persistent database with JSON validation and schema modeling.
  • Execute and verify all CRUD operations with proper HTTP status codes (201 Created, 200 OK, 404 Not Found, 400 Bad Request).
  • Deploy the web service on a public cloud hosting platform (Render) with HTTPS security and environment variable management.
  • Validate both local and live public cloud endpoints using Postman and document request-response payloads.

2. Theoretical Background & REST Principles

REST (Representational State Transfer) is an architectural style for distributed hypermedia systems introduced by Roy Fielding. A RESTful API organizes system functionality around distinct resources, accessible via standardized Uniform Resource Identifiers (URIs).

3. System Architecture & Tech Stack

The REST API is engineered as a lightweight, non-blocking asynchronous server using Node.js and Express.js. Incoming HTTP requests pass through CORS and JSON body-parser middleware, are dispatched by the Express Router to controller functions, and interact with the database before emitting structured JSON responses.

  • Runtime Environment: Node.js (v20+ LTS) | Web Framework: Express.js (v5.0 / v4.x)
  • Database: MongoDB Atlas (Cloud NoSQL) & Supabase (PostgreSQL Cloud)
  • API Testing Client: Postman Desktop Agent & Newman CLI
  • Cloud Hosting Provider: Render Cloud Application Services (Zero-config automated Git deployments)

4.1 Database Schema & Data Model (recordModel.js)

// models/recordModel.js - Database Schema and Data Access Layer
const mongoose = require('mongoose');

const RecordSchema = new mongoose.Schema({
    title: { type: String, required: [true, 'Record title is mandatory'], trim: true },
    category: { type: String, required: true, default: 'General' },
    status: { type: String, enum: ['Active', 'Pending', 'Archived'], default: 'Active' },
    value: { type: Number, required: true },
    description: { type: String, default: '' },
    createdAt: { type: Date, default: Date.now },
    updatedAt: { type: Date, default: Date.now }
});

module.exports = mongoose.model('Record', RecordSchema);

4.2 REST Controller with CRUD Operations (recordController.js)

// controllers/recordController.js - CRUD Business Logic
const Record = require('../models/recordModel');

// 1. CREATE a new record
exports.createRecord = async (req, res) => {
    try {
        const { title, category, status, value, description } = req.body;
        if (!title || value === undefined) {
            return res.status(400).json({ success: false, message: 'Missing required fields: title and value' });
        }
        const newRecord = await Record.create({ title, category, status, value, description });
        return res.status(201).json({ success: true, message: 'Record created successfully', data: newRecord });
    } catch (err) {
        return res.status(500).json({ success: false, error: err.message });
    }
};

// 2. READ all records
exports.getAllRecords = async (req, res) => {
    try {
        const records = await Record.find().sort({ createdAt: -1 });
        return res.status(200).json({ success: true, count: records.length, data: records });
    } catch (err) {
        return res.status(500).json({ success: false, error: err.message });
    }
};

// 3. READ single record by ID
exports.getRecordById = async (req, res) => {
    try {
        const record = await Record.findById(req.params.id);
        if (!record) return res.status(404).json({ success: false, message: 'Record not found with given ID' });
        return res.status(200).json({ success: true, data: record });
    } catch (err) {
        return res.status(400).json({ success: false, message: 'Invalid Record ID format', error: err.message });
    }
};

// 4. UPDATE existing record
exports.updateRecord = async (req, res) => {
    try {
        req.body.updatedAt = new Date();
        const updated = await Record.findByIdAndUpdate(req.params.id, req.body, { new: true, runValidators: true });
        if (!updated) return res.status(404).json({ success: false, message: 'Cannot update: Record not found' });
        return res.status(200).json({ success: true, message: 'Record updated successfully', data: updated });
    } catch (err) {
        return res.status(500).json({ success: false, error: err.message });
    }
};

// 5. DELETE record
exports.deleteRecord = async (req, res) => {
    try {
        const deleted = await Record.findByIdAndDelete(req.params.id);
        if (!deleted) return res.status(404).json({ success: false, message: 'Cannot delete: Record not found' });
        return res.status(200).json({ success: true, message: 'Record deleted successfully', deletedId: req.params.id });
    } catch (err) {
        return res.status(500).json({ success: false, error: err.message });
    }
};

4.3 Routes & Main Express Server (server.js)

// server.js - Express Server Entry Point
const express = require('express');
const cors = require('cors');
const recordController = require('./controllers/recordController');

const app = express();
app.use(cors());
app.use(express.json());

// API Endpoints Mapping
app.get('/api/health', (req, res) => res.json({ status: 'healthy', timestamp: new Date().toISOString() }));
app.post('/api/records', recordController.createRecord);
app.get('/api/records', recordController.getAllRecords);
app.get('/api/records/:id', recordController.getRecordById);
app.put('/api/records/:id', recordController.updateRecord);
app.delete('/api/records/:id', recordController.deleteRecord);

const PORT = process.env.PORT || 5000;
app.listen(PORT, () => console.log(`REST API Server active on port ${PORT}`));
