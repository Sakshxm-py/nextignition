2. Theoretical Background

NoSQL Databases are non-relational database systems designed to store and manage data that is not structured in tabular form. Unlike traditional SQL databases, NoSQL systems offer flexible schemas, horizontal scalability, and high performance for large-scale unstructured data.

MongoDB is the most widely adopted document-oriented NoSQL database. Data is stored as BSON (Binary JSON) documents inside Collections. Each document can have a distinct structure, making MongoDB ideal for agile development.

ConceptMongoDB (NoSQL)SQL (Relational DB)
Data unitDocument (BSON / JSON)Row / Record (Tuple)
Storage groupCollectionTable
SchemaFlexible / DynamicRigid / Fixed
RelationshipsEmbedded or ReferencedForeign Keys / Joins
Query languageMongoDB Query Language (MQL)Structured Query Language (SQL)
Horizontal scalingNative ShardingComplex / Limited

Mongoose is an Object Document Mapper (ODM) for Node.js that provides a schema-based solution to model MongoDB data, with built-in type casting, validation, query building, and middleware hooks.

4.2 Mongoose Schema with Validation (studentModel.js)

// models/studentModel.js
const mongoose = require('mongoose');

const studentSchema = new mongoose.Schema({
    name: {
        type: String,
        required: [true, 'Student name is required'],
        trim: true,
        minlength: [2, 'Name must be at least 2 characters'],
        maxlength: [100, 'Name cannot exceed 100 characters']
    },
    usn: {
        type: String,
        required: [true, 'USN is required'],
        unique: true,
        uppercase: true,
        trim: true,
        match: [/^[A-Z0-9]{6,12}$/, 'USN must be 6-12 alphanumeric characters']
    },
    email: {
        type: String,
        required: [true, 'Email is required'],
        unique: true,
        lowercase: true,
        match: [/^\S+@\S+\.\S+$/, 'Please enter a valid email']
    },
    branch: {
        type: String,
        required: [true, 'Branch is required'],
        enum: { values: ['CSE','ISE','ECE','EEE','ME','Civil','AI-ML','Data Science'], message: '{VALUE} is not a valid branch' }
    },
    semester: {
        type: Number,
        required: [true, 'Semester is required'],
        min: [1, 'Semester must be 1-8'], max: [8, 'Semester must be 1-8']
    },
    cgpa: { type: Number, min: 0, max: 10, default: 0 },
    isActive: { type: Boolean, default: true }
}, { timestamps: true });   // auto adds createdAt & updatedAt

module.exports = mongoose.model('Student', studentSchema);

4.3 CRUD Controller with Error Handling (studentController.js)

// controllers/studentController.js
const Student = require('../models/studentModel');

// CREATE  POST /api/students
exports.createStudent = async (req, res) => {
    try {
        const student = await Student.create(req.body);
        res.status(201).json({ success: true, message: 'Student record created', data: student });
    } catch (err) {
        if (err.code === 11000) {
            const field = Object.keys(err.keyValue)[0];
            return res.status(400).json({ success: false, message: `Duplicate: ${field} already exists` });
        }
        if (err.name === 'ValidationError') {
            const messages = Object.values(err.errors).map(e => e.message);
            return res.status(400).json({ success: false, message: 'Validation failed', errors: messages });
        }
        res.status(500).json({ success: false, error: err.message });
    }
};

// READ ALL  GET /api/students
exports.getAllStudents = async (req, res) => {
    try {
        const { branch, semester, sort = '-createdAt' } = req.query;
        const filter = {};
        if (branch) filter.branch = branch;
        if (semester) filter.semester = Number(semester);
        const students = await Student.find(filter).sort(sort);
        res.status(200).json({ success: true, count: students.length, data: students });
    } catch (err) { res.status(500).json({ success: false, error: err.message }); }
};

// READ ONE  GET /api/students/:id
exports.getStudentById = async (req, res) => {
    try {
        const student = await Student.findById(req.params.id);
        if (!student) return res.status(404).json({ success: false, message: 'Student not found' });
        res.status(200).json({ success: true, data: student });
    } catch (err) { res.status(400).json({ success: false, message: 'Invalid ID', error: err.message }); }
};

// UPDATE  PUT /api/students/:id
exports.updateStudent = async (req, res) => {
    try {
        const student = await Student.findByIdAndUpdate(req.params.id, req.body, { new: true, runValidators: true });
        if (!student) return res.status(404).json({ success: false, message: 'Student not found' });
        res.status(200).json({ success: true, message: 'Student record updated', data: student });
    } catch (err) {
        if (err.name === 'ValidationError') {
            const messages = Object.values(err.errors).map(e => e.message);
            return res.status(400).json({ success: false, errors: messages });
        }
        res.status(500).json({ success: false, error: err.message });
    }
};

// DELETE  DELETE /api/students/:id
exports.deleteStudent = async (req, res) => {
    try {
        const student = await Student.findByIdAndDelete(req.params.id);
        if (!student) return res.status(404).json({ success: false, message: 'Student not found' });
        res.status(200).json({ success: true, message: 'Student deleted permanently', deletedId: req.params.id });
    } catch (err) { res.status(500).json({ success: false, error: err.message }); }
};

4.4 Express Server Entry Point (server.js)

// server.js
const express = require('express');
const cors = require('cors');
const connectDB = require('./db');
const studentController = require('./controllers/studentController');
require('dotenv').config();

const app = express();
app.use(cors());
app.use(express.json());

connectDB();   // Connect to MongoDB Atlas

app.get('/api/health', (req, res) => res.json({ status: 'healthy', db: 'MongoDB Atlas', ts: new Date() }));
app.post('/api/students',     studentController.createStudent);
app.get('/api/students',      studentController.getAllStudents);
app.get('/api/students/:id',  studentController.getStudentById);
app.put('/api/students/:id',  studentController.updateStudent);
app.delete('/api/students/:id', studentController.deleteStudent);

app.use((req, res) => res.status(404).json({ success: false, message: `Route ${req.originalUrl} not found` }));

app.listen(process.env.PORT || 5000, () => console.log('Server running'));
